name: Blender stair sample

on:
  pull_request:
    paths:
      - 'examples/blender_stair_sample/**'
      - '.github/workflows/blender-stair-sample.yml'

permissions:
  contents: read

jobs:
  isolated-sample:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v4
      - name: Install pinned official Godot 4.6.3
        run: |
          curl --fail --location --retry 3 \
            https://github.com/godotengine/godot-builds/releases/download/4.6.3-stable/Godot_v4.6.3-stable_linux.x86_64.zip \
            --output "$RUNNER_TEMP/godot.zip"
          echo "d0bc2113065e481c9c2c2b2c37daa4e8be3fe9e27f0ab9ab0b6096e9a37907f3  $RUNNER_TEMP/godot.zip" | sha256sum --check
          unzip -q "$RUNNER_TEMP/godot.zip" -d "$RUNNER_TEMP/godot"
          sudo install -m 755 "$RUNNER_TEMP/godot/Godot_v4.6.3-stable_linux.x86_64" /usr/local/bin/godot
          godot --version
      - name: Import isolated GLB project
        run: |
          godot --headless --editor --path examples/blender_stair_sample/preview --import --quit 2>&1 | tee "$RUNNER_TEMP/stair-import.log"
          ! grep -E 'SCRIPT ERROR:|Parse Error:|ERROR:' "$RUNNER_TEMP/stair-import.log"
      - name: Test 259 geometry and play checks, input, camera, and beam clipping
        run: |
          for suite in test_blender_sample test_blender_sample_camera_input test_endless_stair_beam; do
            godot --headless --path examples/blender_stair_sample/preview --script "res://tests/$suite.gd" 2>&1 | tee "$RUNNER_TEMP/$suite.log"
            ! grep -E 'SCRIPT ERROR:|Parse Error:|ERROR:' "$RUNNER_TEMP/$suite.log"
          done
